function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. قراءة ورقة "بيانات" لعمل قاموس (Map) للأقسام والنوع بناءً على الرقم الشخصي var sheetBayanat = ss.getSheetByName("بيانات"); var dataBayanat = sheetBayanat.getDataRange().getDisplayValues(); var headersBayanat = dataBayanat[0]; var cprIndex = headersBayanat.indexOf("CPR"); var subjectIndex = headersBayanat.indexOf("Subject"); var typeIndex = headersBayanat.indexOf("النوع"); var bayanatMap = {}; for (var i = 1; i < dataBayanat.length; i++) { var cpr = dataBayanat[i][cprIndex]; if (cpr && cpr.trim() !== "") { bayanatMap[cpr.trim()] = { "Subject": subjectIndex > -1 ? dataBayanat[i][subjectIndex] : "", "النوع": typeIndex > -1 ? dataBayanat[i][typeIndex] : "" }; } } // 2. قراءة ورقة "Form Responses 1" ودمج البيانات var sheetForm = ss.getSheetByName("Form Responses 1"); var dataForm = sheetForm.getDataRange().getDisplayValues(); var headersForm = dataForm[0]; var jsonData = []; for (var j = 1; j < dataForm.length; j++) { var row = dataForm[j]; var obj = {}; var currentCpr = ""; // تحويل الصف إلى كائن (Object) for (var k = 0; k < headersForm.length; k++) { var headerName = headersForm[k].trim(); obj[headerName] = row[k]; if (headerName === "الرقم الشخصي") { currentCpr = row[k].trim(); } } // إذا كان الصف يحتوي على رقم شخصي، نقوم بدمج البيانات if (currentCpr !== "") { if (bayanatMap[currentCpr]) { // أخذ المادة والنوع من ورقة "بيانات" obj["Subject"] = bayanatMap[currentCpr]["Subject"]; obj["النوع_المرتب"] = bayanatMap[currentCpr]["النوع"]; } else { // في حال لم يضاف المعلم لورقة بيانات، نضع قيماً افتراضية obj["Subject"] = "غير محدد"; obj["النوع_المرتب"] = obj["نوع التعاقد"] === "دائم" ? "بحريني" : "وافد"; } jsonData.push(obj); } } return ContentService.createTextOutput(JSON.stringify(jsonData)) .setMimeType(ContentService.MimeType.JSON); }
